import glob
import os
import pyarrow.parquet as pq

files = sorted(glob.glob("data/raw/yellow/*.parquet"))

print("Number of files:", len(files))
print()

total_rows = 0
total_size = 0

for f in files:
    try:
        pf = pq.ParquetFile(f)
        rows = pf.metadata.num_rows
        size = os.path.getsize(f)

        total_rows += rows
        total_size += size

        print(
            f"{os.path.basename(f):40} "
            f"{rows:>12,} rows   "
            f"{size / 1024**2:>7.1f} MB"
        )
    except Exception as e:
        print("ERROR:", f, e)

print()
print("Total files:", len(files))
print(f"Total rows : {total_rows:,}")
print(f"Total size : {total_size / 1024**3:.2f} GiB")
