import re

times = {}

with open("results/ray_run.log") as f:
    for line in f:
        m = re.search(
            r"Completed (\d{4}-\d{2}): .*? ([0-9.]+) seconds",
            line
        )
        if m:
            month, seconds = m.groups()
            times[month] = float(seconds)

print("Months found:", len(times))

for month in sorted(times):
    print(f"{month}: {times[month]:.2f} s")

total = sum(times.values())

print("\nReconstructed full Ray processing time:")
print(f"{total:.2f} seconds")
print(f"{total/60:.2f} minutes")

with open("results/ray_reconstructed_time.txt", "w") as f:
    f.write(f"{total:.6f}\n")
